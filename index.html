$("fixOptions").innerHTML = ""; return; } let w = validWords[Math.floor(Math.random() * validWords.length)]; let correctSentence = (w.ec || "").replace(/[。，！？]/g, "").trim(); // Tạo câu hỏi sai trật tự bằng cách tách cụm/từ rồi xáo trộn let tokens = [];
